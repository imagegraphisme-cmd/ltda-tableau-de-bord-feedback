const FOLDER_ID = "1GbtbIeXh84qbcMgyE4jiL0EH3ckVLRFq"; function doGet() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var configSheet = ss.getSheetByName("Config"); var configData = configSheet.getDataRange().getValues(); var config = { impacts: getColumnData(configData, 0), states: getColumnData(configData, 1), priorities: getColumnData(configData, 2), breakpoints: getColumnData(configData, 3) }; var projectSheet = ss.getSheetByName("Projects"); var projectRows = projectSheet.getDataRange().getValues().slice(1); var projects = projectRows.map(function(row) { return { client: row[0], name: row[1], composition: row[2] ? row[2].split(',').map(function(item) { return item.trim(); }) : [] }; }); var bugSheet = ss.getSheetByName("Bugs"); var bugData = bugSheet.getDataRange().getValues(); var bugs = []; if (bugData.length > 1) { var headers = bugData[0]; for (var i = 1; i < bugData.length; i++) { var row = bugData[i]; var obj = {}; for (var j = 0; j < headers.length; j++) { var val = row[j]; if (val instanceof Date) { if (j === 1) { obj[headers[j]] = Utilities.formatDate(val, Session.getScriptTimeZone(), "dd/MM HH:mm"); } else { obj[headers[j]] = Utilities.formatDate(val, Session.getScriptTimeZone(), "dd/MM/yyyy"); } } else { obj[headers[j]] = val; } } bugs.push(obj); } } bugs.reverse(); return ContentService.createTextOutput(JSON.stringify({ config: config, projects: projects, bugs: bugs })).setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var data = JSON.parse(e.postData.contents); var sheet = ss.getSheetByName("Bugs"); // LOGIQUE LOG BUG if (data.action === "log_bug") { var imageUrl = "No Image"; if (data.imageBlob) { try { var folder = DriveApp.getFolderById(FOLDER_ID); var blob = Utilities.newBlob(Utilities.base64Decode(data.imageBlob.split(",")[1]), data.imageType, data.imageName); imageUrl = folder.createFile(blob).getUrl(); } catch (err) { imageUrl = "Error Image"; } } var lastRow = sheet.getLastRow(); var nextId = "LTDA-" + (lastRow).toString().padStart(3, '0'); // On utilise EXACTEMENT les clés envoyées par l'interface (Majuscules) sheet.appendRow([ nextId, // A: ID new Date(), // B: Timestamp data.dateNoticed || "", // C: Date Noticed "", // D: Comm Note data.client || "", // E: Client data.Project || "", // F: Project data['Noticed By'] || "", // G: Noticed By data.Breakpoint || "", // H: Breakpoint "Not solved", // I: State data.Impact || "", // J: Impact data.Priority || "", // K: Priority data.Location || "", // L: Location data.Note || "", // M: Note imageUrl // N: Screenshot URL ]); return ContentService.createTextOutput(JSON.stringify({status: "success", id: nextId})).setMimeType(ContentService.MimeType.JSON); } // LOGIQUE EDIT BUG if (data.action === "edit_bug") { var rows = sheet.getDataRange().getValues(); for (var i = 1; i < rows.length; i++) { if (rows[i][0] === data.id) { var rowNum = i + 1; sheet.getRange(rowNum, 6).setValue(data.Project); sheet.getRange(rowNum, 7).setValue(data['Noticed By']); sheet.getRange(rowNum, 8).setValue(data.Breakpoint); sheet.getRange(rowNum, 10).setValue(data.Impact); sheet.getRange(rowNum, 11).setValue(data.Priority); sheet.getRange(rowNum, 12).setValue(data.Location); sheet.getRange(rowNum, 13).setValue(data.Note); break; } } return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON); } // LOGIQUE UPDATE STATE if (data.action === "update_bug") { var vals = sheet.getDataRange().getValues(); for (var i = 1; i < vals.length; i++) { if (vals[i][0] === data.id) { sheet.getRange(i + 1, 9).setValue(data.newState); break; } } return ContentService.createTextOutput("Success"); } // LOGIQUE DELETE if (data.action === "delete_bug") { var vals = sheet.getDataRange().getValues(); for (var i = 1; i < vals.length; i++) { if (vals[i][0] === data.id) { sheet.deleteRow(i + 1); break; } } return ContentService.createTextOutput("Success"); } } function getColumnData(data, colIndex) { return data.map(function(row) { return row[colIndex]; }).filter(function(val, i) { return i > 0 && val !== ""; }); }
